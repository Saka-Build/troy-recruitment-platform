    INSERT INTO statuses (
        name,
        colour_hex,
        sort_order
    )
    VALUES (
        'Applied',

        1
    )
    RETURNING id
)

    name,
    colour_hex,
    sort_order

) AS sub(name, colour_hex, sort_order)
